% 3D block in the style of PlotNeuralNet: \cuboid{x}{y}{width}{height}{depth}{colour}
\newcommand{\cuboid}[6]{%
  \begin{scope}[shift={(#1,#2)}, line join=round]
    \filldraw[fill=#6!32!white, draw=black!55, line width=0.3pt] (0,0) -- (#3,0) -- (#3,#4) -- (0,#4) -- cycle;
    \filldraw[fill=#6!16!white, draw=black!55, line width=0.3pt]
      (0,#4) -- (#3,#4) -- ({#3+#5*0.55},{#4+#5*0.45}) -- ({#5*0.55},{#4+#5*0.45}) -- cycle;
    \filldraw[fill=#6!58!white, draw=black!55, line width=0.3pt]
      (#3,0) -- ({#3+#5*0.55},{#5*0.45}) -- ({#3+#5*0.55},{#4+#5*0.45}) -- (#3,#4) -- cycle;
  \end{scope}}
% a snippet of code drawn as bars: \codeglyph{x}{y}{colour of lines}{index of the edited line, 0 = none}{colour of edit}
\newcommand{\codeglyph}[5]{%
  \begin{scope}[shift={(#1,#2)}]
    \filldraw[fill=white, draw=black!55, line width=0.4pt, rounded corners=1pt] (0,0) rectangle (1.35,1.15);
    \foreach \i/\w in {1/0.9,2/0.6,3/1.0,4/0.75,5/0.5}{
      \pgfmathsetmacro{\yy}{1.15-0.2*\i}
      \ifnum\i=#4
        \fill[#5, rounded corners=0.5pt] (0.1,{\yy-0.045}) rectangle ({0.1+\w*1.1},{\yy+0.045});
      \else
        \fill[#3!55!white, rounded corners=0.5pt] (0.1,{\yy-0.035}) rectangle ({0.1+\w*1.1},{\yy+0.035});
      \fi}
  \end{scope}}
